"""Strict new-source stream parser; only a minted authored admission is enabled."""
import csv
import hashlib
import io
from pathlib import Path
import stat
import time
import zipfile
from ..common import canonical_hash,sha256_file
from .national_budget import Blocked
from .f_actual_admission_v1 import require_admission
from .full_adult_types_v1_2 import AdultMatrix,parse_person,i64,FLAGS


def identity(path):
    p=Path(path);s=p.stat()
    return {'device':s.st_dev,'inode':s.st_ino,'bytes':s.st_size,'mtime_ns':s.st_mtime_ns,'ctime_ns':s.st_ctime_ns}


def inspect_source(a):
    require_admission(a);m=a.manifest;c=a.config;path=a.path
    before=identity(path)
    if a.origin=='ACTUAL' and before!=m['expected_identity']:raise Blocked('current source differs from freshly admitted stat identity')
    if path.is_symlink() or before['bytes']!=m['archive_bytes'] or sha256_file(path)!=m['archive_sha256']:raise Blocked('source archive drift')
    if m['state']!='56' or m['period']!=c['period'] or m['metadata_hashes']!=c['metadata_hashes']:raise Blocked('source release/state binding')
    expected={x['name']:x for x in m['members']}
    with zipfile.ZipFile(path) as z:
        infos=z.infolist();names=[i.filename for i in infos]
        if len(infos)!=2 or len(set(n.casefold() for n in names))!=2 or set(names)!={m['person'],m['companion']} or set(expected)!=set(names):raise Blocked('exact two-member policy')
        if sum(x.file_size for x in infos)>c['limits']['state_uncompressed_bytes']:raise Blocked('declared expansion')
        for x in infos:
            e=expected[x.filename]
            if Path(x.filename).name!=x.filename or '\\' in x.filename or stat.S_ISLNK(x.external_attr>>16) or x.flag_bits&1 or x.file_size!=e['bytes'] or x.compress_size!=e['compressed_bytes'] or x.CRC!=e['crc32']:raise Blocked('member identity/path/CRC/encryption')
        comp=expected[m['companion']]
        if comp['bytes']>c['limits']['companion_bytes']:raise Blocked('companion limit')
        with z.open(m['companion']) as f:
            h=hashlib.sha256();n=0
            while True:
                chunk=f.read(min(65536,comp['bytes']-n+1))
                if not chunk:break
                n+=len(chunk);h.update(chunk)
                if n>comp['bytes']:raise Blocked('companion delivered bound')
        if n!=comp['bytes'] or h.hexdigest()!=comp['sha256']:raise Blocked('companion hash')
        with z.open(m['person']) as f:header=f.readline(c['limits']['line_bytes']+1)
        if len(header)!=m['header_bytes'] or hashlib.sha256(header).hexdigest()!=m['header_sha256'] or len(header)>c['limits']['line_bytes']:raise Blocked('header bytes/hash')
        fields=next(csv.reader([header.decode('utf-8-sig')],strict=True))
        if len(fields)!=285 or len(set(fields))!=285 or not set(c['required_fields']).issubset(fields):raise Blocked('literal header/projection')
    if identity(path)!=before:raise Blocked('inspection post-read identity')
    return {'identity':before,'manifest_sha256':canonical_hash(m),'archive_sha256':m['archive_sha256'],'validation_expanded_bytes':len(header)+n,'columns':len(fields),'person_rows_read':0}


def parse_source(a,*,clock=time.monotonic,expected_identity=None):
    require_admission(a);m=a.manifest;c=a.config;start=clock();pre=inspect_source(a);matrix=AdultMatrix();n=0;nb=0;h=hashlib.sha256();state_index=c['states'].index('56')
    if expected_identity is not None and pre['identity']!=expected_identity:raise Blocked('source differs from enrolled file identity')
    def empty():return {'count':0,'point':0,'replicates':[0]*80}
    accounting={k:empty() for k in ('all','adult','under18')};domains={str(d):{k:empty() for k in ('all','adult','under18')} for d in range(6)}
    with zipfile.ZipFile(a.path) as z:
        with z.open(m['person']) as f:
            def lines():
                nonlocal nb
                while True:
                    if clock()-start>c['limits']['state_processing_seconds']:raise Blocked('stream deadline')
                    raw=f.readline(c['limits']['line_bytes']+1)
                    if not raw:break
                    nb+=len(raw);h.update(raw)
                    if len(raw)>c['limits']['line_bytes'] or nb>c['limits']['state_uncompressed_bytes']:raise Blocked('delivered line/CSV expansion')
                    yield raw.decode('utf-8-sig' if nb==len(raw) else 'utf-8')
            reader=csv.DictReader(lines(),strict=True)
            for row in reader:
                n+=1
                if n>c['limits']['state_rows'] or None in row or len(row)!=285 or any(type(v) is not str for v in row.values()):raise Blocked('all-column record width')
                local,w,reps,flags=parse_person(row,'56',c['required_fields']);domain=(0 if int(row['CIT'])<=4 else 1)*3+(0 if int(row['RELSHIPP'])<=36 else int(row['RELSHIPP'])-36)
                which='under18' if local is None else 'adult'
                for target in (accounting['all'],accounting[which],domains[str(domain)]['all'],domains[str(domain)][which]):
                    target['count']+=1;target['point']=i64(target['point']+w)
                    for r,v in enumerate(reps):target['replicates'][r]=i64(target['replicates'][r]+v)
                if local is not None:matrix.add(state_index*72+local,w,reps,flags)
    person=next(e for e in m['members'] if e['name']==m['person'])
    if n!=m['expected_rows'] or nb!=person['bytes'] or h.hexdigest()!=person['sha256']:raise Blocked('actual member rows/bytes/hash')
    post=inspect_source(a)
    if pre!=post:raise Blocked('parse post-read identity/hash')
    matrix.mark_state(state_index)
    part={'schema':'E02_F_WY_'+a.origin+'_STATE_PART_V1','origin':a.origin,'actual_person_rows_processed':n if a.origin=='ACTUAL' else 0,'state':'56','source_manifest_sha256':canonical_hash(m),'source_identity':pre['identity'],'rows':n,'csv_bytes':nb,'expanded_bytes':nb+2*pre['validation_expanded_bytes'],'accounting':accounting,'domains':domains,'primitive_bundle':{k:v.hex() for k,v in matrix.blobs().items()}}
    validate_part(part,a);return part


def validate_part(part,a):
    require_admission(a);m=a.manifest
    expected={'schema','origin','actual_person_rows_processed','state','source_manifest_sha256','source_identity','rows','csv_bytes','expanded_bytes','accounting','domains','primitive_bundle'}
    if type(part) is not dict or set(part)!=expected or part['state']!='56':raise Blocked('closed typed state part')
    person=next(x for x in m['members'] if x['name']==m['person']);comp=next(x for x in m['members'] if x['name']==m['companion'])
    if type(part['rows']) is not int or part['csv_bytes']!=person['bytes'] or part['expanded_bytes']!=person['bytes']+2*(m['header_bytes']+comp['bytes']):raise Blocked('state row/byte identity')
    for group in [part['accounting'],*part['domains'].values()]:
        if set(group)!={'all','adult','under18'}:raise Blocked('accounting projection')
        for account in group.values():
            if set(account)!={'count','point','replicates'} or type(account['count']) is not int or account['count']<0 or type(account['point']) is not int or not account['count']<=account['point']<=9999*account['count'] or type(account['replicates']) is not list or len(account['replicates'])!=80:raise Blocked('accounting integer/domain schema')
            i64(account['point'])
            for value in account['replicates']:i64(value)
    if part['schema']!='E02_F_WY_'+a.origin+'_STATE_PART_V1' or part['origin']!=a.origin or part['actual_person_rows_processed']!=(part['rows'] if a.origin=='ACTUAL' else 0) or part['source_manifest_sha256']!=canonical_hash(m) or part['rows']!=m['expected_rows']:raise Blocked('state part origin/source')
    spec=a.config['spec'];matrix=AdultMatrix.decode({k:bytes.fromhex(v) for k,v in part['primitive_bundle'].items()},spec)
    s=a.config['states'].index('56');expected=bytes([0]*(s*72)+[1]*72+[0]*((50-s)*72))
    if bytes(matrix.present)!=expected:raise Blocked('partial masks must be72known/3600unknown')
    ac=part['accounting'];ds=part['domains']
    if set(ds)!=set(map(str,range(6))) or set(ac)!={'all','adult','under18'}:raise Blocked('six-domain accounting')
    for group in [ac,*ds.values()]:
        for field in ('count','point'):
            if group['all'][field]!=group['adult'][field]+group['under18'][field]:raise Blocked('all-age selection identity')
        if any(group['all']['replicates'][r]!=group['adult']['replicates'][r]+group['under18']['replicates'][r] for r in range(80)):raise Blocked('same-index child identity')
    for group in ('all','adult','under18'):
        for field in ('count','point'):
            if ac[group][field]!=sum(d[group][field] for d in ds.values()):raise Blocked('six-domain conservation')
        if ac[group]['replicates']!=[sum(d[group]['replicates'][r] for d in ds.values()) for r in range(80)]:raise Blocked('six-domain replicas')
    if ac['all']['count']!=part['rows'] or sum(matrix.data['count.i64le'])!=ac['adult']['count'] or sum(matrix.data['point.i64le'])!=ac['adult']['point']:raise Blocked('adult cell identity')
    if ac['adult']['replicates']!=[sum(matrix.data['replicates.i64le'][i*80+r] for i in range(s*72,(s+1)*72)) for r in range(80)]:raise Blocked('adult same-index replicas')
    return matrix
