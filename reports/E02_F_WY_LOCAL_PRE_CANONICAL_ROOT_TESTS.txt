def rows(full=False):
    out=[]
    for age in ([18,25,45,65] if full else [18]):
        for edu in ([17,20,24] if full else [17]):
            for d in range(6):
                row={'RT':'P','STATE':'56','AGEP':str(age),'CIT':'1' if d<3 else '5','RELSHIPP':str([20,37,38][d%3]),'SCHL':str(edu),'PWGTP':str(d+1),'SERIALNO':'PRIVATE_AUTHORED_WY_ONLY'}
                for i in range(80):row['PWGTP'+str(i+1)]=str([-1,3][i%2] if d==0 else d+1)
                for f in ('FAGEP','FCITP','FSCHLP','FRELSHIPP'):row[f]='0'
                out.append(row)
    for d in range(6):
        row=dict(out[d]);row.update(AGEP='2',SCHL='',PWGTP='1')
        for i in range(80):row['PWGTP'+str(i+1)]='1'
        out.append(row)
    return out
